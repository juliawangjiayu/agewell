# 照护协同 Demo Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 一个上帝视角网页三人群聊 demo（公网 demo link + 预录视频）：女佣观察→分级/相关方/差异化输出，雇主指令→拆解/逐项确认，三张常驻 profile 卡片，一键"载入 skill vs 通用好助手"对比，语音输入经 MERaLiON。

**Architecture:** 单 agent 按发信人角色路由到两条流，共用 Railway Postgres 上下文库；判断层是两份 skill（护城河），由 orchestrator 注入 DeepSeek-V4（`temperature=0`）；语音输入先经 MERaLiON STT 转文字再进判断层；`skill_on` 开关控制载入 skill 还是"通用好助手"中性指令。前端 React（CodeBuddy 生成）部署 Vercel，后端 FastAPI + Postgres 部署 Railway，push 到 main 自动上线。

**Tech Stack:** Python 3.11 / FastAPI / **psycopg 3 + Railway Postgres** / pytest / pydantic v2 / **DeepSeek-V4**（OpenAI 兼容，判断层，temp=0）/ **MERaLiON**（api.meralion.ai，语音 STT）/ React（CodeBuddy）/ Vercel + Railway。

## Global Constraints

- **判断层是护城河**：两份 skill 的判断规则（分级/相关方/拆解）不外包，由懂照护/临床的人把关；本计划把 skill 编写与评测排在最早可测处（Task 4/5）。
- **skill 用评测而非单测**：判断层非确定性，用 `backend/evals/` 断言 `grade` 与 `notify`；确定性代码（DB/orchestrator 解析/API/router）走标准 TDD + Mock。
- **判断稳定性**：推理调用一律 `temperature=0`。提交/录制前 `evals` 女佣 4/4 必须在 temp=0 下反复稳过。
- **对比模式诚实性**：`skill_on=True/False` 调**同一模型、同一输入**，唯一变量是 system prompt（skill vs "通用好助手"中性指令）。**baseline 是"通用好助手"，不是"翻译助手"**。
- **语音**：MERaLiON 只做"音频→文字"（女佣/雇主输入），转出的文字进现成文字链路；老人端语音输出只进视频，不进代码。Rosa 语音说 English/Singlish（MERaLiON 不支持 Tagalog）。
- **密钥**：DeepSeek/MERaLiON key 只在后端环境变量，永不进前端打包产物。`.env` 已被 `.gitignore` 排除。
- **占位数据**：展示文字用 `docs/demo-seed-story.md` 的具体内容，`[方括号]` 为需人审替换项，代码里保留方括号原文。
- **提交/部署**：每个 Task 末尾提交；push 到 `main` 自动部署（不搞分支）。commit message 中文 `type: 描述`。
- **模型待定项**：DeepSeek/MERaLiON 的 endpoint/key 从环境变量读，换模型只改 env。

```sql
    id SERIAL PRIMARY KEY,

    family_id INT PRIMARY KEY REFERENCES families(id) ON DELETE CASCADE,
    name TEXT, home_country TEXT, mother_tongue TEXT, care_abilities JSONB DEFAULT '[]'
);
CREATE TABLE IF NOT EXISTS observations (
    id SERIAL PRIMARY KEY,
    family_id INT NOT NULL REFERENCES families(id) ON DELETE CASCADE,
    date TEXT, raw_text TEXT, restored_text TEXT, grade TEXT,
    notify JSONB DEFAULT '[]', reason TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);
CREATE TABLE IF NOT EXISTS messages (
    id SERIAL PRIMARY KEY,
    family_id INT NOT NULL REFERENCES families(id) ON DELETE CASCADE,
    sender_role TEXT, content_original TEXT, content_rendered TEXT,
    meta JSONB DEFAULT '{}', created_at TIMESTAMPTZ NOT NULL DEFAULT now()

    id SERIAL PRIMARY KEY,
```

```python
import json

class FamilyRepo:
    def __init__(self, conn):
        self.conn = conn

    def create_family(self, p: dict) -> int:
        fid = self.conn.execute(
            "INSERT INTO families (label) VALUES (%s) RETURNING id",
            (p.get("label", "demo"),)).fetchone()[0]
        e = p["employer"]
        self.conn.execute(
            "INSERT INTO employer_profile (family_id,name,language,relation,work_schedule,notes)"
            " VALUES (%s,%s,%s,%s,%s,%s)",
            (fid, e.get("name"), e.get("language"), e.get("relation"),
             e.get("work_schedule"), e.get("notes")))
        el = p["elder"]
        self.conn.execute(
            "INSERT INTO elder_profile (family_id,name,age,conditions,baseline_notes,medications,followups)"
            " VALUES (%s,%s,%s,%s,%s,%s,%s)",
            (fid, el.get("name"), el.get("age"), json.dumps(el.get("conditions", []), ensure_ascii=False),
             el.get("baseline_notes"), json.dumps(el.get("medications", []), ensure_ascii=False),
             json.dumps(el.get("followups", {}), ensure_ascii=False)))
        c = p["caregiver"]
        self.conn.execute(
            "INSERT INTO caregiver_profile (family_id,name,home_country,mother_tongue,care_abilities)"
            " VALUES (%s,%s,%s,%s,%s)",
            (fid, c.get("name"), c.get("home_country"), c.get("mother_tongue"),
             json.dumps(c.get("care_abilities", []), ensure_ascii=False)))
        return fid

    def get_family(self, fid: int) -> dict:
        emp = self.conn.execute("SELECT name,language,relation,work_schedule,notes "
                                "FROM employer_profile WHERE family_id=%s", (fid,)).fetchone()
        el = self.conn.execute("SELECT name,age,conditions,baseline_notes,medications,followups "
                               "FROM elder_profile WHERE family_id=%s", (fid,)).fetchone()
        cg = self.conn.execute("SELECT name,home_country,mother_tongue,care_abilities "
                               "FROM caregiver_profile WHERE family_id=%s", (fid,)).fetchone()
        if emp is None:
            raise KeyError(fid)
        return {
            "id": fid,
            "employer": dict(zip(["name","language","relation","work_schedule","notes"], emp)),
            "elder": {"name": el[0], "age": el[1], "conditions": el[2],
                      "baseline_notes": el[3], "medications": el[4], "followups": el[5]},
            "caregiver": dict(zip(["name","home_country","mother_tongue","care_abilities"], cg)),
            "recent_observations": self.recent_observations(fid),
            "tasks": self._tasks(fid),
        }

    def add_observation(self, fid, date, raw, restored, grade, notify, reason) -> int:
        return self.conn.execute(
            "INSERT INTO observations (family_id,date,raw_text,restored_text,grade,notify,reason)"
            " VALUES (%s,%s,%s,%s,%s,%s,%s) RETURNING id",
            (fid, date, raw, restored, grade, json.dumps(notify, ensure_ascii=False), reason)
        ).fetchone()[0]

    def recent_observations(self, fid, limit=10) -> list[dict]:
        rows = self.conn.execute(
            "SELECT date,raw_text,restored_text,grade,notify,reason FROM observations "
            "WHERE family_id=%s ORDER BY id DESC LIMIT %s", (fid, limit)).fetchall()
        keys = ["date","raw_text","restored_text","grade","notify","reason"]
        return [dict(zip(keys, r)) for r in rows]

    def set_last_med_change(self, fid, date) -> None:
        self.conn.execute(
            "UPDATE elder_profile SET followups = followups || %s::jsonb WHERE family_id=%s",
            (json.dumps({"last_med_change_date": date}), fid))

    def add_message(self, fid, role, original, rendered, meta=None) -> int:
        return self.conn.execute(
            "INSERT INTO messages (family_id,sender_role,content_original,content_rendered,meta)"
            " VALUES (%s,%s,%s,%s,%s) RETURNING id",
            (fid, role, original, rendered, json.dumps(meta or {}, ensure_ascii=False))
        ).fetchone()[0]

    def add_task(self, fid, from_msg, item, time, detail) -> int:
        return self.conn.execute(
            "INSERT INTO tasks (family_id,from_message_id,item,time,detail) "
            "VALUES (%s,%s,%s,%s,%s) RETURNING id", (fid, from_msg, item, time, detail)
        ).fetchone()[0]

    def confirm_task(self, task_id) -> None:
        self.conn.execute("UPDATE tasks SET confirmed=TRUE WHERE id=%s", (task_id,))

    def _tasks(self, fid) -> list[dict]:
        rows = self.conn.execute("SELECT id,item,time,detail,confirmed FROM tasks "
                                 "WHERE family_id=%s ORDER BY id", (fid,)).fetchall()
        return [dict(zip(["id","item","time","detail","confirmed"], r)) for r in rows]

    def reset_to_seed(self, fid, seed_payload) -> None:
        """删掉这个家庭的运行时数据（观察/消息/任务），profile 保持，重灌种子观察。"""
        for t in ("tasks", "messages", "observations"):
            self.conn.execute(f"DELETE FROM {t} WHERE family_id=%s", (fid,))
        for o in seed_payload.get("observations", []):
            self.add_observation(fid, o["date"], o["raw_text"], o["restored_text"],
                                 o["grade"], o.get("notify", []), o.get("reason", ""))
        if seed_payload.get("last_med_change_date"):
            self.set_last_med_change(fid, seed_payload["last_med_change_date"])
```

```python
import os
from abc import ABC, abstractmethod
import httpx

class LLMClient(ABC):
    @abstractmethod
    def complete(self, system: str, user_text: str) -> str: ...

class DeepSeekClient(LLMClient):
    def __init__(self):
        self.base = os.environ["LLM_BASE_URL"].rstrip("/")
        self.key = os.environ["LLM_API_KEY"]
        self.model = os.environ["LLM_MODEL"]
    def complete(self, system: str, user_text: str) -> str:
        r = httpx.post(f"{self.base}/chat/completions",
            headers={"Authorization": f"Bearer {self.key}"},
            json={"model": self.model, "temperature": 0,
                  "messages": [{"role": "system", "content": system},
                               {"role": "user", "content": user_text}]},
            timeout=60)
        r.raise_for_status()
        return r.json()["choices"][0]["message"]["content"]

class MockLLMClient(LLMClient):
    def __init__(self, responses: dict[str, str]):
        self.responses = responses
    def complete(self, system: str, user_text: str) -> str:
        for k, v in self.responses.items():
            if k != "_default" and k in system:
                return v
        return self.responses.get("_default", "")
```

```python
import base64
import os
from abc import ABC, abstractmethod
import httpx

class STTClient(ABC):
    @abstractmethod
    def transcribe(self, audio: bytes) -> str: ...

class MeralionClient(STTClient):
    def __init__(self):
        self.url = os.environ["MERALION_URL"]
        self.key = os.environ["MERALION_API_KEY"]
    def transcribe(self, audio: bytes) -> str:
        data_uri = "data:audio/wav;base64," + base64.b64encode(audio).decode()
        r = httpx.post(self.url,
            headers={"Authorization": f"Bearer {self.key}", "Content-Type": "application/json"},
            json={"audio_url": data_uri}, timeout=60)
        r.raise_for_status()
        # 字段名以 MERaLiON 实际返回为准；常见为 {"text": "..."}
        return r.json().get("text", "")

class MockSTTClient(STTClient):
    def __init__(self, text: str):
        self._text = text
    def transcribe(self, audio: bytes) -> str:
        return self._text
```

```python
from pathlib import Path
from app.llm import LLMClient

SKILL_DIR = Path(__file__).parent / "skills"
_FILES = {"helper": "helper_skill.md", "employer": "employer_skill.md"}

def load_skill(role: str, skill_on: bool) -> str:
    fname = "neutral.md" if not skill_on else _FILES[role]
    return (SKILL_DIR / fname).read_text(encoding="utf-8")

def build_user_content(ctx: dict, text: str) -> str:
    return ("【家庭上下文】\n" + json.dumps(ctx, ensure_ascii=False, indent=2)
            + "\n\n【本轮输入】\n" + text)

def parse(role: str, skill_on: bool, raw: str) -> dict:
    if not skill_on:
        return {"skill_on": False, "text": raw.strip()}
    try:
        data = json.loads(raw)
    except json.JSONDecodeError:
        return {"skill_on": True, "error": "unparseable", "raw": raw}
    data["skill_on"] = True
    return data

def handle(llm: LLMClient, ctx: dict, role: str, text: str, skill_on: bool = True) -> dict:
    system = load_skill(role, skill_on)
    raw = llm.complete(system, build_user_content(ctx, text))
    return parse(role, skill_on, raw)
```

```python
import copy, sys, yaml
from pathlib import Path
from app import orchestrator as orch

BASE_CTX = {
    "elder": {"name": "[Ah Ma]", "age": 82, "conditions": ["[高血压]", "[2型糖尿病]"],
              "medications": [{"name": "[Amlodipine]", "timing": "早饭后"}]},
    "employer": {"name": "丽珍"}, "caregiver": {"mother_tongue": "Tagalog"},
    "followups": {"clinic": "[Polyclinic]", "last_med_change_date": None},
    "recent_observations": [],
}

def _ctx(ov: dict) -> dict:
    ctx = copy.deepcopy(BASE_CTX)
    for k, v in (ov or {}).items():
        ctx[k] = {**ctx.get(k, {}), **v} if isinstance(v, dict) else v
    return ctx

def run(cases_path: str, llm, role: str = "helper") -> dict:
    cases = yaml.safe_load(Path(cases_path).read_text(encoding="utf-8"))
    rows, passed = [], 0
    for c in cases:
        out = orch.handle(llm, _ctx(c.get("context_overrides", {})), role, c["input"], skill_on=True)
        if role == "employer":
            tasks = out.get("tasks", [])
            items = " ".join(t.get("item", "") for t in tasks)
            ok = len(tasks) >= c["expected_min_tasks"] and all(s in items for s in c["expected_item_contains"])
            rows.append({"name": c["name"], "got": f"{len(tasks)}tasks/{items}", "pass": ok})
        else:
            ok = out.get("grade") == c["expected_grade"] and sorted(out.get("notify", [])) == sorted(c["expected_notify"])
            rows.append({"name": c["name"], "expected": (c["expected_grade"], sorted(c["expected_notify"])),
                         "got": (out.get("grade"), sorted(out.get("notify", []))), "pass": ok})
        passed += ok
    return {"total": len(cases), "passed": passed, "rows": rows}

if __name__ == "__main__":
    from app.llm import DeepSeekClient
    path = sys.argv[1] if len(sys.argv) > 1 else "evals/cases_helper.yaml"
    role = sys.argv[2] if len(sys.argv) > 2 else "helper"
    res = run(path, DeepSeekClient(), role=role)
    for r in res["rows"]:
        print(("✅" if r["pass"] else "❌"), r["name"], "|", r.get("got"))
    print(f'\n通过 {res["passed"]}/{res["total"]}')
```

```python
from fastapi import FastAPI, HTTPException, UploadFile, Form, File
from fastapi.middleware.cors import CORSMiddleware
from pydantic import BaseModel
from app.router import route
from app import orchestrator as orch
from app.seed import SEED_RESET  # Task 8 提供；先给空 dict 亦可

class MessageIn(BaseModel):
    family_id: int; sender: str; text: str | None = None; skill_on: bool = True

class CompareIn(BaseModel):
    family_id: int; sender: str; text: str | None = None

def create_app(repo, llm, stt) -> FastAPI:
    app = FastAPI(title="照护协同 demo")
    app.add_middleware(CORSMiddleware, allow_origins=["*"], allow_methods=["*"], allow_headers=["*"])

    def _judge(fid, sender, text, skill_on):
        """纯判断，不写库。compare 两侧都走这个，保证对比是只读演示。"""
        role = route(sender)
        ctx = repo.get_family(fid)
        return role, orch.handle(llm, ctx, role, text or "", skill_on=skill_on)

    def _judge_and_persist(fid, sender, text, skill_on):
        """/message 走这个：判断 + 持久化。"""
        role, out = _judge(fid, sender, text, skill_on)
        repo.add_message(fid, role, text or "", "")  # 记录发信人这条消息
        if skill_on and role == "helper" and out.get("grade"):
            # observations 只存精简事实（供未来判断上下文）
            repo.add_observation(fid, "now", text or "", out.get("restored_text", ""),
                                 out["grade"], out.get("notify", []), out.get("reason", ""))
            # 完整判断结果（含三段 outputs）进 messages.meta，供刷新后展示卡片
            repo.add_message(fid, "bot", out.get("restored_text", ""), out.get("reason", ""), meta=out)
        if skill_on and role == "employer" and out.get("tasks"):
            mid = repo.add_message(fid, "bot", out.get("understood", ""), out.get("helper_message", ""), meta=out)
            for t in out["tasks"]:
                repo.add_task(fid, mid, t.get("item"), t.get("time"), t.get("detail"))
        return out

    @app.post("/families")
    def create_family(payload: dict):
        return {"family_id": repo.create_family(payload)}

    @app.get("/families/{fid}")
    def get_family(fid: int):
        try:
            return repo.get_family(fid)
        except KeyError:
            raise HTTPException(404, "not found")

    @app.post("/message")
    def message(m: MessageIn):
        return _judge_and_persist(m.family_id, m.sender, m.text, m.skill_on)

    @app.post("/compare")
    def compare(m: CompareIn):
        # 只读：两侧都用 _judge，不写库
        _, with_skill = _judge(m.family_id, m.sender, m.text, True)
        _, without_skill = _judge(m.family_id, m.sender, m.text, False)
        return {"with_skill": with_skill, "without_skill": without_skill}

    @app.post("/message/audio")
    def message_audio(family_id: int = Form(...), sender: str = Form(...),
                      skill_on: bool = Form(True), file: UploadFile = File(...)):
        text = stt.transcribe(file.file.read())
        return _judge_and_persist(family_id, sender, text, skill_on)

    @app.post("/families/{fid}/reset")
    def reset(fid: int):
        repo.reset_to_seed(fid, SEED_RESET)
        return {"ok": True}

    @app.post("/tasks/{tid}/confirm")
    def confirm(tid: int):
        repo.confirm_task(tid)
        return {"ok": True}

    return app
```
